///|
test "registry JSON Feed through unified public API" {
  let feed = @moonfeed.parse("{\"version\":\"https://jsonfeed.org/version/1.1\",\"title\":\"Installed\",\"items\":[{\"id\":\"one\",\"content_text\":\"body\"}]}")
  assert_eq(feed.title, "Installed")
  assert_eq(feed.items.length(), 1)
  assert_eq(feed.items[0].content_text, Some("body"))
}

///|
test "registry RSS and XML enclosure are included" {
  let feed = @moonfeed.parse_rss("<rss version=\"2.0\"><channel><title>RSS</title><link>https://example.org/</link><description>D</description><item><guid>one</guid><description>Text &amp; XML</description><enclosure url=\"audio\" type=\"audio/mpeg\" length=\"42\"/></item></channel></rss>")
  assert_eq(feed.items[0].id, "one")
  assert_eq(feed.items[0].summary, Some("Text & XML"))
  assert_eq(feed.items[0].attachments[0].size_in_bytes, Some(42UL))
}

///|
test "registry Atom Feed and independent prefixed Entry APIs" {
  let xml = "<feed xmlns=\"http://www.w3.org/2005/Atom\"><id>urn:f</id><title>Atom</title><updated>2026-10-01T12:00:00Z</updated><author><name>A</name></author><entry><id>urn:e</id><title>E</title><updated>2026-10-01T12:00:00Z</updated><content>Body</content></entry></feed>"
  assert_eq(@moonfeed.parse_atom(xml).items[0].content_text, Some("Body"))
  match @moonfeed.parse_atom_document(xml) {
    @moonfeed.AtomDocument::FeedDocument(feed) => assert_eq(feed.entries[0].effective_authors[0].name, "A")
    _ => fail("expected Feed Document")
  }
  match @moonfeed.parse_atom_document("<x:entry xmlns:x=\"http://www.w3.org/2005/Atom\" xml:base=\"https://example.org/a/\"><x:id>urn:e</x:id><x:title>E</x:title><x:updated>2026-10-01T12:00:00Z</x:updated><x:author><x:name>A</x:name></x:author><x:link href=\"../article\"/><x:content>Body</x:content></x:entry>") {
    @moonfeed.AtomDocument::EntryDocument(entry) => assert_eq(entry.metadata.links[0].href.resolved, "https://example.org/article")
    _ => fail("expected Entry Document")
  }
}

///|
test "registry format detection dates and errors are public" {
  assert_true(@moonfeed.detect_format("<rss version=\"2.0\"/>") is Rss)
  assert_eq(@moonfeed.parse_datetime("2002-10-02T15:00:00+02:00").unix_seconds, 1033563600L)
  let rejected = try { ignore(@moonfeed.parse("<html/>")); false } catch { @moonfeed.Detect(_) => true; _ => false }
  assert_true(rejected)
}

///|
fn installed_rss(items : String) -> String {
  "<rss version='2.0'><channel><title>Installed RSS</title><link>https://example.org/</link><description>Feed</description>" + items + "</channel></rss>"
}

///|
test "registry 0.3 description-only RSS retains absent identity through public APIs" {
  let xml = installed_rss("<item><description>这是一条新闻正文</description></item>")
  let feed : @moonfeed.Feed = @moonfeed.parse(xml)
  let item : @moonfeed.FeedItem = feed.items[0]
  assert_eq(feed.items.length(), 1)
  assert_eq(item.title, None)
  assert_eq(item.id, "")
  assert_eq(item.url, None)
  assert_eq(item.content_html, Some("这是一条新闻正文"))
  assert_eq(@moonfeed.parse_rss(xml), feed)
  assert_eq(@moonfeed.parse_podcast_rss(xml).feed, feed)
}

///|
test "registry 0.3 title-only and title-description RSS are legal without guid or link" {
  for (fields, expected_body) in [
    ("<title>只有标题的新闻</title>", None),
    ("<title>只有标题的新闻</title><description>新闻正文</description>", Some("新闻正文")),
  ] {
    let item = @moonfeed.parse_rss(installed_rss("<item>" + fields + "</item>")).items[0]
    assert_eq(item.title, Some("只有标题的新闻"))
    assert_eq(item.content_html, expected_body)
    assert_eq(item.id, "")
    assert_eq(item.url, None)
  }
}

///|
test "registry 0.3 missing identities do not discard or reorder multiple items" {
  let items = @moonfeed.parse_rss(installed_rss(
    "<item><description>One</description></item>" +
    "<item><title>Two</title><guid isPermaLink='false'>known</guid></item>" +
    "<item><description>Three</description></item>",
  )).items
  assert_eq(items.length(), 3)
  assert_eq(items.map(item => item.id), ["", "known", ""])
  assert_eq(items[0].content_html, Some("One"))
  assert_eq(items[2].content_html, Some("Three"))
}

///|
test "registry 0.3 XML boundaries reject text roots CDATA and references but accept Misc" {
  let xml = installed_rss("<item><description><![CDATA[Body]]>&amp;</description></item>")
  let expected = @moonfeed.parse_rss(xml)
  assert_eq(expected.items[0].content_html, Some("Body&"))
  for (prefix, suffix) in [
    (" \t\r\n", " \t\r\n"),
    ("<!--before-->", "<!-- valid trailing comment -->"),
    ("<?audit before?>", "<?audit complete?>"),
  ] {
    assert_eq(@moonfeed.parse(prefix + xml + suffix), expected)
  }
  for input in [
    "INVALID" + xml, xml + "INVALID", xml + "<other/>",
    "<![CDATA[]]>" + xml, xml + "<![CDATA[ ]]>",
    "&#32;" + xml, xml + "&#xA;", xml + "<unfinished",
    xml + "<!-- invalid -- comment -->", xml + "<?audit incomplete",
  ] {
    let rejected = try { ignore(@moonfeed.parse(input)); false } catch {
      @moonfeed.Rss(_) | @moonfeed.Detect(_) => true
      _ => false
    }
    assert_true(rejected)
  }
}

///|
test "registry 0.3 JSON attachments and public model types retain concrete values" {
  let feed = @moonfeed.parse_json_feed(
    "{\"version\":\"https://jsonfeed.org/version/1.1\",\"title\":\"Attachments\",\"items\":[{\"id\":\"one\",\"content_text\":\"Body\",\"attachments\":[{\"url\":\"https://example.org/audio.mp3\",\"mime_type\":\"audio/mpeg\",\"title\":\"Audio\",\"size_in_bytes\":42,\"duration_in_seconds\":1.5}]}]}",
  )
  assert_eq(feed.items[0].attachments.length(), 1)
  let attachment : @moonfeed.Attachment = feed.items[0].attachments[0]
  assert_eq(attachment.url, "https://example.org/audio.mp3")
  assert_eq(attachment.mime_type, "audio/mpeg")
  assert_eq(attachment.title, Some("Audio"))
  assert_eq(attachment.size_in_bytes, Some(42UL))
  assert_eq(attachment.duration_in_seconds, Some(1.5))
  assert_true(@moonfeed.detect_format(installed_rss("")) is @moonfeed.FeedFormat::Rss)
  assert_true(@moonfeed.detect_format("{\"version\":\"https://jsonfeed.org/version/1.1\"}") is @moonfeed.FeedFormat::JsonFeed)
  assert_true(@moonfeed.detect_format("<a:feed xmlns:a='http://www.w3.org/2005/Atom'/>") is @moonfeed.FeedFormat::Atom)
}

///|
fn registry_entry(extra : String) -> String {
  "<entry xmlns='http://www.w3.org/2005/Atom'><id>urn:installed</id><title>Installed</title><updated>2026-10-01T12:00:00Z</updated><author><name>A</name></author>" + extra + "<content>Body</content></entry>"
}

///|
test "registry 0.3.1 entity budget rejects empty amplification with a typed error" {
  let mut subset = "<!ENTITY e0 ''>"
  for i = 1; i <= 20; i = i + 1 {
    let reference = "&e" + (i - 1).to_string() + ";"
    subset += "<!ENTITY e" + i.to_string() + " '" + reference + reference + "'>"
  }
  let xml = "<!DOCTYPE rss [" + subset + "]>" + installed_rss("<item><description>&e20;</description></item>")
  assert_true(@test.expect_error(() => @moonfeed.parse(xml)) is @moonfeed.Rss(@rss.InvalidXml("XML entity reference limit exceeded")))
}

///|
test "registry 0.3.1 declaration errors reject NDATA and forward defaults" {
  for (subset, message) in [
    ("<!ENTITY % p SYSTEM 'urn:p' NDATA n>", "NDATA is forbidden for parameter entities"),
    ("<!ATTLIST rss a CDATA '&later;'><!ENTITY later 'v'>", "undeclared general entity"),
  ] {
    match @test.expect_error(() => @moonfeed.parse_rss("<!DOCTYPE rss [" + subset + "]>" + installed_rss("<item><description>Body</description></item>"))) {
      @rss.InvalidXml(actual) => assert_eq(actual, message)
      _ => fail("expected typed XML declaration error")
    }
  }
  let valid = "<!DOCTYPE rss [<!ENTITY a '&b;'><!ENTITY b 'Safe'>]>" + installed_rss("<item><description>&a;</description></item>")
  assert_eq(@moonfeed.parse(valid).items[0].content_html, Some("Safe"))
}

///|
test "registry 0.3.1 related MIME inner type comparison is registration specific" {
  let duplicate = registry_entry("<link href='a' type='multipart/related;boundary=AbC;type=&quot;Text/HTML&quot;'/><link href='b' type='Multipart/Related;TYPE=&quot;text/html&quot;;BOUNDARY=AbC'/>")
  assert_true(@test.expect_error(() => @moonfeed.parse_atom_document(duplicate)) is @atom.InvalidRequiredField("entry.link[1]", _))
  match @moonfeed.parse_atom_document(registry_entry("<link href='a' type='multipart/related;boundary=AbC;type=&quot;text/html&quot;'/><link href='b' type='multipart/related;boundary=abc;type=&quot;text/html&quot;'/>") ) {
    @moonfeed.AtomDocument::EntryDocument(entry) => assert_eq(entry.metadata.links.length(), 2)
    _ => fail("expected Entry Document")
  }
}

///|
test "registry 0.3.1 private charset values retain case distinctions" {
  match @moonfeed.parse_atom_document(registry_entry("<link href='a' type='application/x-private;charset=AbC'/><link href='b' type='application/x-private;CHARSET=abc'/>") ) {
    @moonfeed.AtomDocument::EntryDocument(entry) => {
      assert_eq(entry.metadata.links.length(), 2)
      assert_eq(entry.metadata.links[0].media_type, Some("application/x-private;charset=AbC"))
    }
    _ => fail("expected Entry Document")
  }
}
